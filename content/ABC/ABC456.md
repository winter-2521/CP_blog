#### B
どのさいころに4,5,6のいずれを割り当てるか？を全探索。
```cpp
int a[3][6];
void solve(){
  vint vs = {4,5,6};
  rep(i,3){
    rep(j,6) cin >> a[i][j];
  }
  long double ans = 0;
  do{
    long double now = 1;
    rep(i,3){
      int cnt = 0;
      rep(j,6) cnt += (a[i][j] == vs[i]);
      now *= (long double)cnt/6;
    }
    ans += now;
  }while(next_permutation(ALL(vs)));
  cout << ans << endl;
}
```
#### C
文字列を隣り合う文字が等しい部分で分割すると、答えはそれぞれの文字列の長さを $S$ とすると、$\sum{S*(S+1)/2}$ が答え。
途中まで誤読して D問題を解いており、謎のDP (dpにする必要はない) を書いた。

```cpp
void solve(){
  string s; cin >> s;
  int n = s.size();
  vector<mint> dp(3,0);
  mint ans = 0;
  rep(i,n){
    vector<mint> np(3,0);
    int tar = (int)(s[i]-'a');
    rep(j,3){
      if(j == tar){
        dp[j] = 0;
        np[j] += 1;
      }else{
        np[tar] += dp[j];
      }
    }
    swap(np,dp);
    rep(j,3) ans += dp[j];
  }
  cout << ans.val() << endl;
}
```
#### D

動的計画法で解ける。
`dp[i][c]`:= `i` 文字目まで見て最後の文字が `c` と定義すると、

`dp[i][S_i] = dp[i-1][x]+dp[i-1][y]+1` ※`x,y` は `S_i` と異なる2文字
`dp[i][x] = dp[i-1][x]` ※`x` は `S_i` と異なる文字
と更新でき、答えは `sum(dp[N])` である。
前者の遷移は「自分から文字列を開始する1通り」 + 「自分と異なる文字で終わった文字列とマージ」である。後者は更新なしの現状維持。

```cpp
void solve(){
  string s; cin >> s;
  int n = s.size();
  vector<mint> dp(3,0);
  rep(i,n){
    vector<mint> np = dp;
    int tar = (int)(s[i]-'a');
    np[tar]++;
    rep(j,3)if(j != tar){
      np[tar] += dp[j];
    }
    swap(np,dp);
  }
  mint ans = 0;
  rep(i,3) ans += dp[i];
  cout << ans.val() << endl;
}
```

#### F
なんとなく尺取法で更新していきたくなる。連続 $K$ 項塗れるまで進めてコストを更新、左端を削除を繰り返したいが、削除がうまくいかない。
dpを考えると遷移が行列で表示できそうな形になって、一度applyしてから単体revertみたいなことが難しそうなのでセグ木に乗せた。

https://atcoder.jp/contests/abc456/submissions/75454402

```cpp
using mt = Matrix<ll>;
mt op(mt a,mt b){ return b*a; }
mt e(){ return mt(2,2).unit();}

void solve(){
  int n,k; cin >> n >> k;
  ll ans = INF,now = 0;
  vint a(n); rep(i,n) cin >> a[i];
  if(k == 1){
    cout << *min_element(ALL(a)) << endl;
    return;
  }

  vector<mt> v(n);
  rep(i,n){
    mt m(2,2);
    m[0][0] = a[i];
    m[0][1] = a[i];
    m[1][0] = 0;
    v[i] = m;
  }
  segtree<mt,op,e> seg(v);

  rep(i,n)if(i+k <= n){
    mt l = seg.prod(i+1,i+k);
    chmin(ans,l[0][0]+a[i]);
    if(i+k+1 <= n){
      l = seg.prod(i+1,i+k+1);
      chmin(ans,l[0][0]+a[i]);
    }
  }
  cout << ans << endl;
}
```
